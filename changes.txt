const getApiBaseUrl = () => {
    if (process.env.REACT_APP_API_URL) {
        return process.env.REACT_APP_API_URL;
    }
    const hostname = window.location.hostname || 'localhost';
    const protocol = window.location.protocol || 'http:';
    return `${protocol}//${hostname}:5000`;
};

export const API_BASE_URL = getApiBaseUrl();

# Update-3: CloudSigner Cryptographic Digital Signature & Alignment Fix
--------------------------------------------------------------------------------
1. COMPONENT: backend/server.js & backend/utils/generateInvoicePDF.js
--------------------------------------------------------------------------------
SUMMARY OF CHANGES:
• Eliminated post-signing pdf-lib PDFDocument re-serialization. Re-saving signed
  PDFs with pdf-lib stripped /ByteRange and /Contents, causing Adobe Acrobat error:
  "Signature contains incorrect, unrecognized, corrupted or suspicious data (SigDict /Contents illegal data)".
• Attachments and main invoice are now combined BEFORE signing so the entire document
  is digitally signed in one pass, and the raw signed buffer is uploaded directly to Google Drive.
• Fixed signature stamp vertical squishing by enforcing minimum dimensions (width: 180pt, height: 65pt)
  and expanding the HTML gap between "For : JAYARAMA ASSOCIATES" and "Authorized Signatory" from 65px to 90px.
• Enforced strict CloudSigner payload schema: options: { reason, locations, customtext } and root location: [x, y, w, h].
